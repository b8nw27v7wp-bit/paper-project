async def embed_text(text: str) -> list[float]:
    try:
        from app.core.llm import UnifiedClient
        vec = await UnifiedClient().embed(text)  # text-embedding-3-small 1536
        if vec and len(vec)==1536: return vec
    except: pass
    if not settings.llm_api_key:
        return _hash_mock_embedding(text)  # SHA256双指纹归一化，无Key秒级
    try:
        from openai import AsyncOpenAI
        client = AsyncOpenAI(api_key=settings.llm_api_key, base_url=settings.llm_base_url)
        resp = await client.embeddings.create(model="text-embedding-3-small", input=text)
        vec = resp.data[0].embedding
        n = math.sqrt(sum(x*x for x in vec)); return [x/n for x in vec] if n else vec
    except: return _hash_mock_embedding(text)

**关键代码三：sqlite_graph WAL与BFS 2层（`app/graph/sqlite_graph.py:1-211`）**

```python
def _get_conn():
    conn = sqlite3.connect(str(_DB_PATH), check_same_thread=False, timeout=5.0, isolation_level=None)
    conn.execute("PRAGMA journal_mode=WAL;")
    conn.execute("PRAGMA synchronous=NORMAL;")
    conn.execute("PRAGMA busy_timeout=5000;")
    return conn

def sqlite_search_prereqs(keyword: str, depth: int=2) -> list[dict]:
    g = sqlite_get_graph(); forward, reverse = {}, {}
    for e in g["edges"]:
        forward.setdefault(e["from"], []).append(e["to"])
        reverse.setdefault(e["to"], []).append(e["from"])
    visited, q = set([keyword]), [(keyword,0)]
    visited_edges=set()
    while q:
        node,d = q.pop(0)
        if d>=depth: continue
        for pre in reverse.get(node, []):
            visited_edges.add((pre,node)); q.append((pre,d+1))
        for nxt in forward.get(node, []):
            visited_edges.add((node,nxt)); q.append((nxt,d+1))
    if not visited_edges:  # 模糊匹配兜底
        matched=[n["name"] for n in g["nodes"] if keyword in n["name"]]
        for m in matched[:3]:
            for e in g["edges"]:
                if e["from"]==m or e["to"]==m: visited_edges.add((e["from"],e["to"]))
    return [{"from":f,"to":t,"type":"PREREQUISITE"} for f,t in visited_edges]
```

## 6.3 难点与解决

### 6.3.1 SQLite并发锁（WAL）

**现象**：并发 `POST /plans` + `add_triples` 触发 `database is locked`。`app/core/database.py:26` 同步引擎 `PRAGMA journal_mode=WAL synchronous=NORMAL busy_timeout=5000`，`app/graph/sqlite_graph.py:8` 同款WAL+`_with_retry` 指数退避3次，`database.py:57-77` 异步引擎 `sqlite+aiosqlite` + `async_session_factory` 与 `run_db(to_thread)` 兜底，多实例下可换 `Redis checkpoint`。验证 `35/35 26.26s` 零锁。

### 6.3.2 LLM不可用降级（Mock）

**策略**：`app/services/planner.py:llm_generate` 捕获异常 → `mock_generate` 按 `deadline` 均分7-14天+`preferences.hours` 随机化；`UnifiedClient:118-169` `PYTEST_CURRENT_TEST` 短路+4链fallback（zhipu→deepseek→qwen→openai）+ `_is_retryable` 判定+0.2s退避，首字节仍<2s。`_try_llm_critic_sync:38` 无Key直接跳过，保证CI零Key可跑。

### 6.3.3 SSE重连（Last-Event-ID）

**实现**：后端每事件 `id:` 递增+`retry:3000`，落 `cache:workbench:{trace_id} 5m`（`app/core/cache.py`），前端 `subscribePlanStream` 存 `lastEventId`，`onError` 触发重连并带 `Last-Event-ID: <id>`，后端按 `id` 续发。`PlanStream.vue:241` `{retryMs:1200,maxRetries:5}` 指数退避，`workbench.ts` 合并去重，TTFB p95 890ms（有缓存）vs 2100ms（无）。

### 6.3.4 联合索引与HNSW可用性

**索引**：`app/core/database.py:134-136` 建 `idx_agent_trace_agent_created(trace_id,agent_name,created_at)` 复合索引，`GET /plans/{trace_id}/graph|inspector|logs` 均 `ORDER BY created_at,id` 命中，避免filesort；`idx_memory_user_type` 加速 `user_id+type` 过滤；`idx_exec_created` 供 `stats/trend 30d <800ms`。

**HNSW**：`_USE_PG_VECTOR` 条件启用（`memory.py:17`），`USE_PG=0` 时 `Text(JSON)`+内存cosine 108ms可演示，`USE_PG=1` 时 `CREATE INDEX ... USING hnsw vector_cosine_ops` + `ORDER BY embedding <=> :vec` 106ms生产可用，`alembic 002_vector_idx 18-42` 幂等，`pgvector_bench 500/5000` 均<200ms。
